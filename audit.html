comp_summary <- comp |>
  dplyr::group_by(source) |>
  dplyr::summarise(
    n_vars     = dplyr::n(),
    pct_min    = round(min(pct_munis), 1),
    pct_median = round(median(pct_munis), 1),
    pct_max    = round(max(pct_munis), 1),
    n_zero     = sum(pct_munis == 0),
    n_full     = sum(pct_munis == 100),
    .groups = "drop"
  ) |>
  dplyr::arrange(desc(pct_median))

DT::datatable(
  comp_summary,
  colnames = c("Source", "# Variables", "Min %", "Median %", "Max %",

  rownames = FALSE,

  comp <- readr::read_csv(completeness_path, show_col_types = FALSE)

  comp <- tibble::tibble(variable = character(), n_munis_with_data = integer(),

}
