NoteAbout this panel

This panel explores relationships between variables rather than the distribution of any single variable. Each section asks a different question:

  • §2.1 Scatterplots: Are conflict history and justice infrastructure correlated? Do municipalities with more peacebuilding investment have lower violence? Points are municipality means across all available years. Axes are log-scaled to handle extreme skew.
  • §2.2 Bivariate Moran’s I: Does the geographic neighbor of a high-conflict municipality also have low justice infrastructure? Bivariate spatial autocorrelation tests this.
  • §2.3 LISA Maps: Which specific municipalities drive the spatial clustering? High-High = conflict cluster; Low-Low = peaceful cluster; High-Low / Low-High = spatial outliers (interesting cases).
  • §2.4 MAUP Sensitivity: Does the spatial pattern depend on how we allocate UCDP events to municipal boundaries? This section tests robustness to different aggregation rules.

All spatial weights use Queen contiguity restricted to the 170 PDET municipalities.

2.2 Bivariate Moran’s I

Show code 
compute_bivariate_moran <- function(x, y, lw) {
  x[is.na(x)] <- 0
  y[is.na(y)] <- 0
  if (stats::var(x) == 0 || stats::var(y) == 0) return(c(I = NA, p = NA))
  x_std <- as.numeric(scale(x))
  y_std <- as.numeric(scale(y))
  wy    <- spdep::lag.listw(lw, y_std, zero.policy = TRUE)
  I     <- mean(x_std * wy)
  perm_I <- replicate(999, {
    yp  <- y_std[sample(length(y_std))]
    wyp <- spdep::lag.listw(lw, yp, zero.policy = TRUE)
    mean(x_std * wyp)
  })
  p <- mean(abs(perm_I) >= abs(I))
  c(I = I, p = p)
}

if (!is.null(lw_q) && !is.null(pdet_munis) && nrow(panel) > 0) {

```{r setup}
# ── Helper: aggregate a panel variable to cross-section (mean across years) ──
panel_cross <- function(var_col) {
  if (!var_col %in% names(panel)) return(NULL)
  panel |>
    dplyr::group_by(dane_code) |>
    dplyr::summarise(v = mean(.data[[var_col]], na.rm = TRUE), .groups = "drop")
}

# ── Spatial weights (Queen contiguity, PDET-only) ──
lw_queen <- NULL
get_lw <- function() {
  if (is.null(lw_queen) && !is.null(pdet_munis)) {
    pdet_sp <- sf::as_Spatial(pdet_munis)
    nb <- spdep::poly2nb(pdet_sp, queen = TRUE)
    lw_queen <<- spdep::nb2listw(nb, style = "W", zero.policy = TRUE)
  }
  lw_queen
}
```

```{r scatter-tabs}
if (nrow(panel) > 0) {
  for (pair in CORE_VARS) {

      dplyr::filter(!is.na(x_val) & !is.na(y_val))
    if (nrow(df_join) < 5) {
      cat("> Fewer than 5 matched rows.\n\n")
      next
    }
    rho <- cor(log1p(df_join$x_val), log1p(df_join$y_val),
               method = "spearman", use = "complete.obs")
    p <- plotly::plot_ly(
      df_join,
      x = ~log1p(x_val), y = ~log1p(y_val),
      type = "scatter", mode = "markers",
      text = ~paste0("<b>", muni_name, "</b>",
                     "<br>", pair$x_label, ": ", round(x_val, 1),
                     "<br>", pair$y_label, ": ", round(y_val, 1)),
      hoverinfo = "text",
      marker = list(size = 7, opacity = 0.65, color = "#2166ac")
    ) |>

          x = 0.05, y = 0.95, xref = "paper", yref = "paper",
```

## 2.3 LISA Maps

::: panel-tabset
```{r lisa-maps-p2}
#| results: asis
LISA_COLOURS <- c("HH" = "#b2182b", "LL" = "#2166ac",
                  "HL" = "#f4a582", "LH" = "#92c5de",
                  "Not significant" = "#dddddd")

lisa_vars_p2 <- list(
  list(var = "V_conflict_events_all__ucdp",     label = "UCDP Events"),
  list(var = "V_selective_killing_cases__cnmh",  label = "CNMH Selective Killings"),
  list(var = "V_homicides_mean__jep",            label = "Homicide Rate (VERDATA)"),
  list(var = "J_n_justice_rooms__slj",           label = "Justice Rooms (SLJ)")
)

lw_q <- get_lw()

if (!is.null(lw_q) && !is.null(pdet_munis) && nrow(panel) > 0) {
  geo_base <- pdet_munis |> sf::st_drop_geometry() |> dplyr::select(dane_code)

  for (lv in lisa_vars_p2) {
    cat(glue::glue("\n\n### {lv$label}\n\n"))
    vd <- panel_cross(lv$var)
    if (is.null(vd)) {
      cat("> Variable not in panel.\n\n")
      next
    }
    vals <- dplyr::left_join(geo_base, vd, by = "dane_code") |> dplyr::pull(v)
    vals[is.na(vals)] <- 0

    if (stats::var(vals) == 0) {
      cat("> Zero variance — skipping LISA.\n\n")
      next
    }
    locm <- tryCatch(
      suppressWarnings(spdep::localmoran(vals, lw_q, zero.policy = TRUE)),
      error = function(e) NULL)
    if (is.null(locm)) {
      cat("> LISA computation failed.\n\n")
      next
    }
    vals_std <- as.numeric(scale(vals))
    lag_vals <- spdep::lag.listw(lw_q, vals_std, zero.policy = TRUE)
    quad <- dplyr::case_when(
      vals_std > 0 & lag_vals > 0 & locm[, 5] < 0.05 ~ "HH",
      vals_std < 0 & lag_vals < 0 & locm[, 5] < 0.05 ~ "LL",
      vals_std > 0 & lag_vals < 0 & locm[, 5] < 0.05 ~ "HL",
      vals_std < 0 & lag_vals > 0 & locm[, 5] < 0.05 ~ "LH",
      TRUE ~ "Not significant"
    )
    lisa_sf <- pdet_munis |>
      dplyr::mutate(lisa_quad = quad,
                    lisa_I    = locm[, 1],
                    lisa_p    = locm[, 5])

    lisa_sf$lisa_quad <- factor(lisa_sf$lisa_quad,
                                 levels = c("HH","LL","HL","LH","Not significant"))
    g <- ggplot2::ggplot(lisa_sf) +
      ggplot2::geom_sf(ggplot2::aes(fill = lisa_quad),
                       color = "grey30", linewidth = 0.2) +
      ggplot2::scale_fill_manual(values = LISA_COLOURS, drop = FALSE,
                                  name = "LISA (p<0.05)") +
      ggplot2::labs(title = lv$label) +
      ggplot2::theme_void(base_size = 10) +
      ggplot2::theme(legend.position = "right")
    print(g)
    cat("\n\n")
  }
} else {
  cat("> Panel or geometry not loaded.\n")
}
```
